이미지 자료 수정하기


from skimage import io, color
import matplotlib.pyplot as plt
import numpy as np

# 외부 이미지 불러오기
url = "주소"
image = io.imread(url)  # 이미지 로드

print(type(image))
print(image.shape)

# 원본 이미지 출력
plt.figure(figsize=(10, 5))
plt.imshow(image)
plt.title("Original Image")


# RGB filter
red_only = np.zeros_like(image)
green_only = np.zeros_like(image)
blue_only = np.zeros_like(image)

red_only[..., 0] = image[..., 0]
green_only[..., 1] = image[..., 1]
blue_only[..., 2] = image[..., 2]

#원색 결합 이미지

plt.imshow(red_only+green_only)

# 원형 마스크
rows, cols, _ = image.shape
center = (rows // 2, cols // 2)
radius = min(rows, cols) // 3
Y, X = np.ogrid[:rows, :cols]
mask = (X - center[1])**2 + (Y - center[0])**2 <= radius**2

masked_image = np.zeros_like(image)
masked_image[mask] = image[mask]


# 흑백 이미지 출력
image_gray = color.rgb2gray(image)  # 흑백(그레이스케일) 변환
plt.figure(figsize=(10, 5))
plt.imshow(image_gray, cmap='gray')
plt.title("Original Grayscale Image")
plt.axis("off")
print(image_gray.shape)
print(np.max(image_gray), np.min(image_gray))


#흑백반전수행
image_inverted = 1 - image_gray
plt.imshow(image_inverted, cmap='gray')


# 그림 일부만 가져오기
partial_image = image[0:140, 300:440,:]

plt.figure(figsize=(2, 2))
plt.imshow(partial_image)
plt.title("Partial Image")
plt.axis("off")
plt.show()



#모자이크

mosaic_image = image[::20, ::20]
plt.figure(figsize=(5, 5))
plt.imshow(mosaic_image)
plt.title("Image")
plt.axis("off")
plt.show()
